class Solution {
public:
    int compress(vector<char>& chars) {
        int ansindex=0;
        int i=0 ;
        int n=chars.size();
        while(i<n)
        {
            int j=i+1 ;
            chars[ansindex++]=chars[i] ;
            while (j<n && chars[i]==chars[j])
                j++ ;
            
            int diff = j-i ;
            if (diff>1)
            {
                string s = to_string(diff) ;
                for (char ch : s)
                {
                    chars[ansindex++] = ch ;
                }
            }
            
            i=j ;
        }
        while (chars.size()>ansindex)
            chars.pop_back() ;
        return ansindex ;
    }
};
